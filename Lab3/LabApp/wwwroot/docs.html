Что смотреть

Свои структуры построены на одном примитиве — односвязном списке. Стек и очередь — тонкие адаптеры над ним, плюс адаптеры над стандартными коллекциями .NET для сравнения в разделе Benchmarks.

Связный список

Односвязный список с указателями на первый и последний узел. Хранит счётчик элементов, поэтому Count за O(1). Указатель на хвост (Tail) даёт O(1) на добавление в конец. Удаления из конца O(n) — обратного указателя нет.

Класс SinglyLinkedList

namespace LabApp.Core;

public sealed class SinglyLinkedList<T>
{
    public ListNode<T>? Head { get; private set; }
    public ListNode<T>? Tail { get; private set; }
    public int Count { get; private set; }
    public bool IsEmpty => Head is null;

    public void AddFirst(T value)
    {
        var node = new ListNode<T>(value) { Next = Head };
        Head = node;
        Tail ??= node;
        Count++;
    }

    public void AddLast(T value)
    {
        var node = new ListNode<T>(value);
        if (Tail is null) { Head = Tail = node; }
        else { Tail.Next = node; Tail = node; }
        Count++;
    }

    public T RemoveFirst()
    {
        if (Head is null) throw new InvalidOperationException("Список пуст");
        var v = Head.Value;
        Head = Head.Next;
        if (Head is null) Tail = null;
        Count--;
        return v;
    }

    public T RemoveLast()
    {
        if (Head is null) throw new InvalidOperationException("Список пуст");
        if (Head == Tail) { var v0 = Head.Value; Head = Tail = null; Count--; return v0; }
        var cur = Head;
        while (cur!.Next != Tail) cur = cur.Next;
        var v = Tail!.Value;
        cur.Next = null;
        Tail = cur;
        Count--;
        return v;
    }

    public void SetHeadOnly(ListNode<T>? head) => Head = head;

    public void RecomputeTailAndCount()
    {
        Tail = null;
        int c = 0;
        for (var cur = Head; cur is not null; cur = cur.Next) { Tail = cur; c++; }
        Count = c;
    }

    public IEnumerable<T> Items
    {
        get
        {
            for (var cur = Head; cur is not null; cur = cur.Next) yield return cur.Value;
        }
    }

    public SinglyLinkedList<T> Clone()
    {
        var copy = new SinglyLinkedList<T>();
        foreach (var v in Items) copy.AddLast(v);
        return copy;
    }

    public override string ToString() => "[" + string.Join(", ", Items) + "]";
}

Стек

Интерфейс IStack<T> — контракт для обеих реализаций. LinkedListStack<T> использует наш односвязный список: Push = AddFirst, Pop = RemoveFirst. StandardStackAdapter<T> оборачивает System.Collections.Generic.Stack<T> и нужен, чтобы в разделе Benchmarks обе реализации шли через один и тот же интерфейс.

Интерфейс и реализации

namespace LabApp.Core;

public interface IStack<T>
{
    int Count { get; }
    void Push(T value);
    T Pop();
    T Top();
    bool IsEmpty();
    IReadOnlyList<T> Print();
}

public sealed class LinkedListStack<T> : IStack<T>
{
    private readonly SinglyLinkedList<T> _inner = new();
    public int Count => _inner.Count;
    public void Push(T value) => _inner.AddFirst(value);
    public T Pop() => _inner.RemoveFirst();
    public T Top()
    {
        var head = _inner.Head;
        if (head is null) throw new InvalidOperationException("Стек пуст");
        return head.Value;
    }
    public bool IsEmpty() => _inner.IsEmpty;
    public IReadOnlyList<T> Print() => _inner.Items.ToList();
}

public sealed class StandardStackAdapter<T> : IStack<T>
{
    private readonly System.Collections.Generic.Stack<T> _inner = new();
    public int Count => _inner.Count;
    public void Push(T value) => _inner.Push(value);
    public T Pop() => _inner.Pop();
    public T Top() => _inner.Peek();
    public bool IsEmpty() => _inner.Count == 0;
    public IReadOnlyList<T> Print() => _inner.ToList();
}

Почему обе реализации идут через IStack<T> в бенчмарках: так интерфейсная диспетчеризация одинакова, и разница во времени отражает именно разницу структур, а не разницу API. См. ADR-0005.

Очередь

Интерфейс IQueue<T>, две реализации по той же схеме. ListQueue<T> использует наш список: Enqueue = AddLast, Dequeue = RemoveFirst. Оба O(1), потому что список хранит и Head, и Tail. StandardQueueAdapter<T> оборачивает System.Collections.Generic.Queue<T> (ring buffer внутри).

Интерфейс и реализации

namespace LabApp.Core;

public interface IQueue<T>
{
    int Count { get; }
    void Enqueue(T value);
    T Dequeue();
    T Peek();
    bool IsEmpty();
    IReadOnlyList<T> Print();
}

public sealed class ListQueue<T> : IQueue<T>
{
    private readonly SinglyLinkedList<T> _inner = new();
    public int Count => _inner.Count;
    public void Enqueue(T value) => _inner.AddLast(value);
    public T Dequeue() => _inner.RemoveFirst();
    public T Peek()
    {
        var head = _inner.Head;
        if (head is null) throw new InvalidOperationException("Очередь пуста");
        return head.Value;
    }
    public bool IsEmpty() => _inner.IsEmpty;
    public IReadOnlyList<T> Print() => _inner.Items.ToList();
}

public sealed class StandardQueueAdapter<T> : IQueue<T>
{
    private readonly System.Collections.Generic.Queue<T> _inner = new();
    public int Count => _inner.Count;
    public void Enqueue(T value) => _inner.Enqueue(value);
    public T Dequeue() => _inner.Dequeue();
    public T Peek() => _inner.Peek();
    public bool IsEmpty() => _inner.Count == 0;
    public IReadOnlyList<T> Print() => _inner.ToList();
}
